import numpy as np
import os

class Node:
    def __init__(self, node_no, data, parent, act, cost):
        self.data = data
        self.parent = parent
        self.act = act
        self.node_no = node_no
        self.cost = cost

def get_initial():
    initial_state = np.zeros(9)
    for i in range(9):
        initial_state[i] = int(input())
    return np.reshape(initial_state, (3,3))

def find_index(puzzle):
    i, j = np.where(puzzle == 0)
    return int(i), int(j)

def move_left(data):
    i, j = find_index(data)
    if j == 0:
        return None
    temp = np.copy(data)
    temp[i][j], temp[i][j-1] = temp[i][j-1], temp[i][j]
    return temp

def move_right(data):
    i, j = find_index(data)
    if j == 2:
        return None
    temp = np.copy(data)
    temp[i][j], temp[i][j+1] = temp[i][j+1], temp[i][j]
    return temp

def move_up(data):
    i, j = find_index(data)
    if i == 0:
        return None
    temp = np.copy(data)
    temp[i][j], temp[i-1][j] = temp[i-1][j], temp[i][j]
    return temp

def move_down(data):
    i, j = find_index(data)
    if i == 2:
        return None
    temp = np.copy(data)
    temp[i][j], temp[i+1][j] = temp[i+1][j], temp[i][j]
    return temp

def move_tile(action, data):
    if action == 'up': return move_up(data)
    if action == 'down': return move_down(data)
    if action == 'left': return move_left(data)
    if action == 'right': return move_right(data)

def path(node):
    p = []
    while node:
        p.append(node)
        node = node.parent
    return p[::-1]

def bfs(root):
    goal = [[1,2,3],[4,5,6],[7,8,0]]
    q = [root]
    visited = []

    while q:
        cur = q.pop(0)
        if cur.data.tolist() == goal:
            return cur
        for m in ['up','down','left','right']:
            temp = move_tile(m, cur.data)
            if temp is not None and temp.tolist() not in visited:
                visited.append(temp.tolist())
                q.append(Node(0,temp,cur,m,0))

k = get_initial()

n=int(input())
board=[[0]*n for _ in range(n)]

def isSafe(r,c):
    for i in range(n):
        if board[r][i]==1 or board[i][c]==1:
            return False
    i,j=r-1,c-1
    while i>=0 and j>=0:
        if board[i][j]==1:
            return False
        i-=1;j-=1
    i,j=r-1,c+1
    while i>=0 and j<n:
        if board[i][j]==1:
            return False
        i-=1;j+=1
    return True

def solve(count):
    if count==n:
        return True
    for i in range(n):
        for j in range(n):
            if isSafe(i,j):
                board[i][j]=1
                if solve(count+1):
                    return True
                board[i][j]=0
    return False

solve(0)

    print(*r)
